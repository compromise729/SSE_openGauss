## 编译安装SSE_openGauss

**环境：**

centos7.6

**第三方库：**[https://opengauss.obs.cn-south-1.myhuaweicloud.com/2.0.0/openGauss-third_party_binarylibs.tar.gz](https://opengauss.obs.cn-south-1.myhuaweicloud.com/2.0.0/openGauss-third_party_binarylibs.tar.gz)

**前置环境：**

groupadd dbgrp -g 2000

useradd omm -g 2000 -u 2000

passwd omm

mkdir -p {/opt/og,/opt/ogdata}

cd  /opt/og/



yum install -y git

curl -s [https://packagecloud.io/install/repositories/github/git-lfs/script.rpm.sh](https://packagecloud.io/install/repositories/github/git-lfs/script.rpm.sh) | sudo bash

sudo yum install git-lfs

git lfs install

git clone [https://github.com/hareOuO/SSE_openGauss](https://github.com/hareOuO/SSE_openGauss) openGauss-server

cd openGauss-server

git submodule update --init --recursive

yum install -y libaio-devel-0.3.109

yum install -y flex 2.5.37

 yum install -y bison 2.7-4

yum install -y ncurses-devel 5.9-13.20130511

yum install -y glibc-devel 2.17-111

yum install -y patch 2.7.1-10

yum -y install redhat-lsb 4.1

yum install -y readline-devel 7.0-13



tar -zxvf openGauss-third_party_binarylibs.tar.gz

mv openGauss-third_party_binarylibs binarylibs

cd binarylibs/buildtools/centos7.6_x86_64/gcc7.3/gcc/bin

rm -rf /usr/bin/gcc

ln -s /opt/og/binarylibs/buildtools/centos7.6_x86_64/gcc7.3/gcc/bin/gcc   /usr/bin/gcc

yum install -y python3.6*

rm -rf  /usr/bin/python

mv /usr/bin/python3.6 /usr/bin/python

python3.6安装完成后还需要修改下两个文件的文件头.分别执行vim /usr/libexec/urlgrabber-ext-down和vim /usr/bin/yum,将这两个文件的第一行:#!/usr/bin/python修改为#!/usr/bin/python2.7 即可



chown -R omm: {/opt/og,/opt/ogdata}

chmod -R 755 /opt/og

chmod -R 700 /opt/ogdata

su - omm



vi ~/.bashrc



export CODE_BASE=/opt/og/openGauss-server

export BINARYLIBS=/opt/og/binarylibs

export GAUSSHOME=$CODE_BASE/dest/

export GCC_PATH=$BINARYLIBS/buildtools/centos7.6_x86_64/gcc7.3

export CC=$GCC_PATH/gcc/bin/gcc

export CXX=$GCC_PATH/gcc/bin/g++

export LD_LIBRARY_PATH=$GAUSSHOME/lib:$GCC_PATH/gcc/lib64:$GCC_PATH/isl/lib:$GCC_PATH/mpc/lib/:$GCC_PATH/mpfr/lib/:$GCC_PATH/gmp/lib/:$LD_LIBRARY_PATH

export PATH=$GAUSSHOME/bin:$GCC_PATH/gcc/bin:$PATH



source ~/.bashrc



./configure --gcc-version=7.3.0 CC=g++ CFLAGS='-O0' --prefix=$GAUSSHOME --3rd=$BINARYLIBS --enable-debug --enable-cassert --enable-thread-safety --without-readline --without-zlib

**cmake3.29.3(源码安装)：**

  sudo yum groupinstall "Development Tools"

  sudo yum install openssl-devel  

  wget https://github.com/Kitware/CMake/releases/download/v3.29.3/cmake-3.29.3.tar.gz  

  tar -zxvf cmake-3.29.3.tar.gz  

  cd cmake-3.29.3  

   ./bootstrap  

   make  

   sudo make install  

   cmake --version 

**安装sodium：**

sudo yum install epel-release

sudo yum install libsodium-devel


**安装relic：**

git clone https://github.com/relic-toolkit/relic.git 

cd relic 

mkdir build && cd build  

cmake -DCMAKE_INSTALL_PREFIX=/opt/og/fides_relic ..  

make 

sudo make install  


git clone https://github.com/relic-toolkit/relic.git mitra_relic

cd mitra_relic/

git checkout 94154a8b77e13d7fd24c9081e7fc6cd95956473d

mkdir build && cd build/ 

cmake -DCMAKE_POSITION_INDEPENDENT_CODE=ON -DCMAKE_INSTALL_PREFIX=/opt/og/
mitra_relic ..  

make

make install


**安装gmp：**

 sudo yum install gmp gmp-devel -y  


**安装openssl：**

yum install openssl-devel

**安装scons：**

sudo pip3 install scons  

**安装boost：**

wget https://sourceforge.net/projects/boost/files/boost/1.64.0/boost_1_64_0.tar.gz

tar -xzvf boost_1_64_0.tar.gz

cd boost_1_64_0/

./bootstrap.sh --prefix=/usr/local

./b2

sudo ./b2 install

**安装yasm：**

yum install epel-release

yum install yasm

**装crypto++：**

 sudo yum install cryptopp cryptopp-devel  

 rpm -qi cryptopp  

**编译安装：**

make -j4

make install

**启动：**

gs_initdb --pgdata=/opt/ogdata/ --nodename=dn_6001 --encoding=utf-8

export LD_LIBRARY_PATH=/opt/og/openGauss-server/sse/opensse-schemes/third_party/crypto/build/src:$LD_LIBRARY_PATH

export LD_LIBRARY_PATH=/opt/og/openGauss-server/sse/Mitra/third_party/crypto/library/lib:$LD_LIBRARY_PATH

gs_ctl start -D /opt/ogdata -Z single_node -l logfile

gsql -d postgres -p 5432
